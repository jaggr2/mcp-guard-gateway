  if (!row) return errorResult(`Approval not found: ${approvalId}`);
  switch (row.status) {
    case 'approved': return runApproved(deps, row);
    case 'denied': return errorResult(`Approval ${approvalId} was denied by the operator`);
    case 'expired': return errorResult(`Approval ${approvalId} expired without approval`);
    case 'pending': {
      if (isExpired(row)) {
        const policy = resolvePolicy(deps.config.policyTable, row.tool); // hmm need policy table in deps
        if (policy.timeout_action === 'approve') {
          await deps.store.setApprovalStatus(row.id, 'approved', { tgUserId: row.tg_user_id });
          const updated = { ...row, status: 'approved' as const };
          return runApproved(deps, updated);
        }
        await deps.store.setApprovalStatus(row.id, 'expired', { tgUserId: row.tg_user_id });
        await recordAudit(deps, row.tool, JSON.parse(row.args), row.reason, 'expired', { expiresAt: row.expires_at });
        return errorResult(`Approval ${approvalId} expired without approval (timeout_action: deny)`);
      }
      return pendingResult(approvalId);
    }
  }
}

export function buildDeps(env: Env, callerId: string, protocolVersion: string | null): Deps {
  const config = loadConfig(env, protocolVersion);
  return {

    store: new D1Store(config.db),

export function namespaceTool(tool, namespace): Tool {
  return { ...tool, name: `${namespace}_${tool.name}` };
}

export function buildMcpServer(deps: Deps): Server {
  const server = new Server({ name: "mcp-guard-gateway", version: "0.1.0" }, { capabilities: { tools: {} } });

  server.setRequestHandler(ListToolsRequestSchema, async (request) => {
    const ns = deps.config.namespace;
    const cursor = request.params?.cursor;
    const { tools, nextCursor } = await deps.upstream.toolsList(cursor);
    const namespaced: Tool[] = tools.map((t) => {
      const tool = { ...t, name: `${ns}_${t.name}` } as Tool;
      const policy = resolvePolicy(deps.policyTable, tool.name);
      if (requiresReason(policy)) {
        tool.inputSchema = injectReasonSchema(tool.inputSchema);
      }
      return tool;
    });
    return {
      tools: [...namespaced, ...GATEWAY_TOOLS],
      ...(nextCursor ? { nextCursor } : {}),
    };
  });

  server.setRequestHandler(CallToolRequestSchema, async (request) => {
    const { name, arguments: args = {} } = request.params;
    return callTool(deps, name, args);
  });

  return server;
}

async function callTool(deps, name, args): Promise<CallToolResult> {
  if (GATEWAY_TOOL_NAMES.has(name)) {
    return callGatewayTool(deps, name, args);
  }
  const prefix = `${deps.config.namespace}_`;
  if (!name.startsWith(prefix)) {
    return errorResult(`Unknown tool: ${name}`);
  }
  const upstreamName = name.slice(prefix.length);
  const reason = extractReason(args);
  const cleanArgs = stripReason(args);
  const policy = resolvePolicy(deps.policyTable, name);

  if (requiresReason(policy)) {
    assertReason(reason); // throws -32602
  }

  if (policy.guard === "hitl") {
    const started = await startApproval(deps, name, cleanArgs, reason!, policy);
    return jsonResult(started);
  }

  let result: CallToolResult;
  try {
    result = await deps.upstream.callTool(upstreamName, cleanArgs);
  } catch (err) {
    result = errorResult(`Upstream execution failed: ${err instanceof Error ? err.message : String(err)}`);
  }
  await recordAudit({ store: deps.store, callerId: deps.callerId }, name, cleanArgs, reason, "auto-audited", {
    upstreamStatus: result.isError ? `error: ${firstText(result)}` : "ok",
  });
  return result;
}
